### 2.2 api/request.js（Axios 封装）

```javascript
import axios from 'axios'

const request = axios.create({
  baseURL: 'http://localhost:8080/api/v1',
  timeout: 10000
})

// 响应拦截：统一取 data
request.interceptors.response.use(
  response => response.data,
  error => {
    console.error('API 请求失败:', error)
    return Promise.reject(error)
  }
)

export default request
```

## 五、页面开发模板

每个 `.vue` 页面都遵循相同结构，以 Dashboard.vue 为例：

```vue
<template>
  <div class="page-container">
    <h2>院长驾驶舱</h2>
    <FilterBar @filter="handleFilter" />

    <!-- KPI 卡片行 -->
    <el-row :gutter="16">
      <el-col :span="4" v-for="kpi in kpiList" :key="kpi.title">
        <KpiCard :title="kpi.title" :value="kpi.value" :trend="kpi.trend" />
      </el-col>
    </el-row>

    <!-- 图表行 -->
    <el-row :gutter="16">
      <el-col :span="12"><div ref="chart1" style="height:350px"></div></el-col>
      <el-col :span="12"><div ref="chart2" style="height:350px"></div></el-col>
    </el-row>

    <el-row :gutter="16">
      <el-col :span="12"><div ref="chart3" style="height:350px"></div></el-col>
      <el-col :span="12"><div ref="chart4" style="height:350px"></div></el-col>
    </el-row>
  </div>
</template>

<script>
import { getDashboardKpi, getDashboardTrend, getDashboardPayment, getDeptRank } from '@/api'
import KpiCard from '@/components/KpiCard'
import FilterBar from '@/components/FilterBar'

export default {
  components: { KpiCard, FilterBar },
  data() {
    return { kpiList: [...] }
  },
  mounted() { this.loadData() },
  methods: {
    async loadData() {
      const kpi = await getDashboardKpi()
      // 填充 kpiList
      this.$nextTick(() => {
        this.initChart1()
        this.initChart2()
      })
    },
    initChart1() {
      const chart = this.$echarts.init(this.$refs.chart1)
      chart.setOption({ xAxis:..., yAxis:..., series:[{type:'line',data:...}] })
    },
    handleFilter({ dateRange, dept }) { this.loadData() }
  }
}
</script>
```
